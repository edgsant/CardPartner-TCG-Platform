	\section{El Concepto Card Partner}
	\begin{frame}{Contexto y Problema}
		\begin{itemize}
			\item \textbf{Crecimiento de los TCG:} el mercado de cartas coleccionables está en auge (One Piece, Lorcana, etc.).
			\item \textbf{Información Dispersa:} los jugadores deben saltar entre 4 webs distintas para precios, mazos, noticias y reglas.
			\item \textbf{Barrera de Entrada:} los principiantes se sienten abrumados por la falta de guías centralizadas.
		\end{itemize}
		\vspace{0.3cm}
		\begin{block}{Nuestra Propuesta}
			Una aplicación web \textbf{All-in-One} que centraliza catálogo, mercado, construcción de mazos y comunidad en una experiencia fluida.
		\end{block}
	\end{frame}
	
	\section{Diferenciación Competitiva}
	\begin{frame}{¿Por qué Card Partner es diferente?}
		A diferencia de sitios como \textit{Card Market} o \textit{Limitless TCG}, nosotros aportamos:
		\vspace{0.3cm}
		\begin{description}
			\item[Enfoque Social Real:] ristema de comentarios y valoraciones para democratizar el "META".
			\item[Gestión de Colección:] \textit{wishlist} integrada y gestión de mazos personales/públicos.
			\item[Inteligencia Contextual:] traducción asistida por IA que entiende la jerga técnica (Ramp, OTK, Mill).
			\item[Visualización Dinámica:] constructor de mazos con filtros inteligentes basados en el líder seleccionado.
		\end{description}
	\end{frame}

			\section{El Corazón: Integración de APIs}
			\begin{frame}{Sincronización Automática (OPTCG API)}
				\textbf{Reto:} Mantener 2000+ cartas actualizadas manualmente es imposible.
				\begin{itemize}
					\item \textbf{Consumo REST:} uso de la librería \textbf{GSON} para parsear JSONs masivos.
					\item \textbf{Artes Alternativos:} algoritmo basado en \texttt{HashMap} para detectar duplicados y asignar sufijos de ID (\texttt{\_p1}, \texttt{\_p2}).
					\item \textbf{Robustez:} implementación de "Escudos Antimisioles" (timeouts y validadores de nulos).
				\end{itemize}
				\begin{exampleblock}{Resultado}
					Base de datos siempre al día con los últimos lanzamientos de Bandai de forma autónoma.
				\end{exampleblock}
			\end{frame}
			
			\begin{frame}[fragile]{Inteligencia Artificial Contextual (Gemini/Groq)}
				\textbf{Diferenciador Clave:} traducción que no parece de Google.
				\begin{itemize}
					\item \textbf{Llama-3 via Groq:} integración mediante HTTP Post.
					\item \textbf{Prompts Dinámicos:} la IA recibe instrucciones distintas según si traduce un mazo (técnico) o un comentario (natural).
					\item \textbf{Tolerancia a Fallos:} si la API no responde, el sistema muestra el texto original sin romper la interfaz.
				\end{itemize}
				\vspace{0.2cm}
				\begin{lstlisting}
					String instruccion = switch (tipo) {
						case "MAZO" -> "Translate using TCG jargon (Ramp, OTK). ONLY output text: ";
						...
					};
				\end{lstlisting}
			\end{frame}
